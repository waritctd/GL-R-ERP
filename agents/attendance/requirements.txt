# showroom_agent.py supports two device transports, chosen by ZK_TRANSPORT:
#   pullsdk (default) - the ZKAccess3.5 Pull SDK (plcommpro.dll) via ctypes only
#                       (stdlib). Requires 32-bit Python on Windows. Showroom SC700.
#   pyzk              - pyzk's standalone ZK protocol. Warehouse ZMM220 (which the
#                       Pull SDK cannot connect to). Any Python bitness; needs pyzk.
#
# Dependencies:
#   requests - posting punches to the ERP.
#   tzdata   - IANA timezone database. Windows ships no system tz data, so
#              zoneinfo.ZoneInfo("Asia/Bangkok") fails without this package.
#   pyzk     - only needed when ZK_TRANSPORT=pyzk (warehouse). Harmless otherwise;
#              it is imported lazily, so the pullsdk path never touches it.
#   pg8000   - cvaccess_agent.py only: reads the showroom SenseFace 3A's punches
#              out of ZKBio CVAccess's bundled PostgreSQL. Pure Python, so it
#              installs on any Windows Python bitness.
requests>=2.31.0
tzdata>=2024.1
pyzk>=0.9
pg8000>=1.31
